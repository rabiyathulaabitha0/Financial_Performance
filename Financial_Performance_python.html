df['Gross_Profit'] = pd.to_numeric(df['Gross_Profit'], errors='coerce')

df['Interest_Rate'] = pd.to_numeric(df['Interest_Rate'], errors='coerce')
df['Write_Off_Amount'] = pd.to_numeric(df['Write_Off_Amount'], errors='coerce')
df['Risk_Score'] = pd.to_numeric(df['Risk_Score'], errors='coerce')
df['Date'] = pd.to_datetime(df['Date'], errors='coerce')

df['Write_Off_Amount'] = df['Write_Off_Amount'].fillna(0)
df['Loan_Amount'] = df['Loan_Amount'].fillna(0)
